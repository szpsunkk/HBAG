```python
# core/constellation.py
import numpy as np
from typing import List, Tuple, Dict, Optional
import ephem  # PyEphem


class Satellite:
    """单颗卫星状态"""
    
    def __init__(self, sat_id: int, orbit_id: int, pos_in_orbit: int,
                 altitude_km: float, inclination_deg: float, n_orbits: int,
                 n_sats_per_orbit: int):
        self.sat_id = sat_id
        self.orbit_id = orbit_id
        self.pos_in_orbit = pos_in_orbit
        self.altitude_km = altitude_km
        self.inclination_deg = inclination_deg
        self.n_orbits = n_orbits
        self.n_sats_per_orbit = n_sats_per_orbit
        
        # 轨道周期 (s)
        R_EARTH_KM = 6371.0
        GM = 398600.4418  # km^3/s^2
        r = R_EARTH_KM + altitude_km
        self.orbital_period_s = 2 * np.pi * np.sqrt(r**3 / GM)
        
        # 当前状态
        self.position_ecef = np.zeros(3)  # ECEF坐标 (km)
        self.illuminated = True           # 光照状态
        self.sun_angle_rad = 0.0          # 太阳角度
        
    def compute_position(self, time_s: float) -> np.ndarray:
        """
        计算卫星在时刻t的ECEF坐标
        基于Walker Delta星座轨道力学
        """
        R_EARTH_KM = 6371.0
        r = R_EARTH_KM + self.altitude_km
        
        # 初始相位角
        raan_0 = 2 * np.pi * self.orbit_id / self.n_orbits  # 升交点赤经
        ma_0 = 2 * np.pi * self.pos_in_orbit / self.n_sats_per_orbit  # 平近点角
        
        # 当前真近点角（假设圆轨道）
        mean_motion = 2 * np.pi / self.orbital_period_s
        true_anomaly = ma_0 + mean_motion * time_s
        
        # 倾角
        inc = np.deg2rad(self.inclination_deg)
        
        # 轨道坐标系到ECI
        x_orb = r * np.cos(true_anomaly)
        y_orb = r * np.sin(true_anomaly)
        
        # 旋转到ECI
        # 考虑地球自转（简化处理）
        earth_rotation = 7.2921150e-5  # rad/s
        theta_earth = earth_rotation * time_s
        
        x_eci = (x_orb * np.cos(raan_0) - y_orb * np.cos(inc) * np.sin(raan_0))
        y_eci = (x_orb * np.sin(raan_0) + y_orb * np.cos(inc) * np.cos(raan_0))
        z_eci = y_orb * np.sin(inc)
        
        # ECI到ECEF
        x_ecef = x_eci * np.cos(theta_earth) + y_eci * np.sin(theta_earth)
        y_ecef = -x_eci * np.sin(theta_earth) + y_eci * np.cos(theta_earth)
        z_ecef = z_eci
        
        self.position_ecef = np.array([x_ecef, y_ecef, z_ecef])
        return self.position_ecef
    
    def compute_illumination(self, time_s: float) -> Tuple[bool, float]:
        """
        计算光照状态和太阳角度
        基于MAAC-IILP [2] 方程(13)中的 phi 和 sigma
        """
        # 太阳方向（简化处理）
        EARTH_ORBIT_PERIOD = 365.25 * 24 * 3600  # s
        sun_angle = 2 * np.pi * time_s / EARTH_ORBIT_PERIOD
        sun_dir = np.array([np.cos(sun_angle), np.sin(sun_angle), 0.0])
        
        # 地影检测
        R_EARTH_KM = 6371.0
        sat_pos = self.position_ecef
        sat_dist = np.linalg.norm(sat_pos)
        
        # 投影到太阳方向
        proj = np.dot(sat_pos, sun_dir)
        
        if proj < 0:  # 卫星在地球背光侧
            # 垂直距离
            perp_dist = np.sqrt(max(sat_dist**2 - proj**2, 0))
            self.illuminated = perp_dist > R_EARTH_KM
        else:
            self.illuminated = True
        
        # 太阳角度（太阳能板法向与阳光夹角）
        if self.illuminated:
            panel_normal = sat_pos / sat_dist  # 假设法向朝向地心
            self.sun_angle_rad = np.arccos(
                np.clip(np.dot(panel_normal, sun_dir), -1, 1)
            )
        else:
            self.sun_angle_rad = 0.0
        
        return self.illuminated, self.sun_angle_rad


class Constellation:
    """
    LEO mega-constellation模型
    基于SATFLOW [1] 的星座架构
    """
    
    def __init__(self, config: dict):
        self.n_orbits = config['n_orbits']
        self.n_sats_per_orbit = config['n_sats_per_orbit']
        self.altitude_km = config['altitude_km']
        self.inclination_deg = config['inclination_deg']
        self.n_satellites = self.n_orbits * self.n_sats_per_orbit
        self.name = config.get('name', 'Unknown')
        
        # 创建卫星
        self.satellites: List[Satellite] = []
        for orbit_id in range(self.n_orbits):
            for pos_id in range(self.n_sats_per_orbit):
                sat_id = orbit_id * self.n_sats_per_orbit + pos_id
                sat = Satellite(
                    sat_id, orbit_id, pos_id,
                    self.altitude_km, self.inclination_deg,
                    self.n_orbits, self.n_sats_per_orbit
                )
                self.satellites.append(sat)
        
        # ISL邻居关系
        self.adjacency: np.ndarray = np.zeros(
            (self.n_satellites, self.n_satellites), dtype=int
        )
        self._build_isl_topology()
        
    def _build_isl_topology(self):
        """
        构建ISL拓扑：每颗卫星有4条ISL（2条轨道内+2条轨道间）
        基于SATFLOW [1] Section III-A
        """
        for orbit_id in range(self.n_orbits):
            for pos_id in range(self.n_sats_per_orbit):
                sat_id = orbit_id * self.n_sats_per_orbit + pos_id
                
                # 轨道内ISL（前后各一颗）
                prev_pos = (pos_id - 1) % self.n_sats_per_orbit
                next_pos = (pos_id + 1) % self.n_sats_per_orbit
                
                prev_id = orbit_id * self.n_sats_per_orbit + prev_pos
                next_id = orbit_id * self.n_sats_per_orbit + next_pos
                
                self.adjacency[sat_id, prev_id] = 1
                self.adjacency[sat_id, next_id] = 1
                
                # 轨道间ISL（东西各一条）
                if orbit_id > 0:
                    west_orbit = orbit_id - 1
                    west_id = west_orbit * self.n_sats_per_orbit + pos_id
                    self.adjacency[sat_id, west_id] = 1
                    
                if orbit_id < self.n_orbits - 1:
                    east_orbit = orbit_id + 1
                    east_id = east_orbit * self.n_sats_per_orbit + pos_id
                    self.adjacency[sat_id, east_id] = 1
    
    def update(self, time_s: float) -> Dict:
        """
        更新所有卫星状态
        返回：{sat_id: {position, illuminated, sun_angle}}
        """
        states = {}
        for sat in self.satellites:
            pos = sat.compute_position(time_s)
            illuminated, sun_angle = sat.compute_illumination(time_s)
            states[sat.sat_id] = {
                'position': pos,
                'illuminated': illuminated,
                'sun_angle': sun_angle,
                'orbit_id': sat.orbit_id,
                'pos_in_orbit': sat.pos_in_orbit
            }
        return states
    
    def compute_distances(self, time_s: float) -> np.ndarray:
        """
        计算所有卫星对之间的欧氏距离 (km)
        """
        positions = np.array([
            sat.compute_position(time_s) for sat in self.satellites
        ])
        M = self.n_satellites
        dist_matrix = np.zeros((M, M))
        
        for i in range(M):
            for j in range(M):
                if i != j:
                    dist_matrix[i, j] = np.linalg.norm(
                        positions[i] - positions[j]
                    )
        return dist_matrix
    
    def get_neighbors(self, sat_id: int) -> List[int]:
        """获取卫星的ISL邻居列表"""
        return list(np.where(self.adjacency[sat_id] == 1)[0])
    
    def get_eclipse_fraction(self, time_s: float) -> float:
        """
        计算当前Eclipse分数 theta
        用于验证Proposition 2 [6]
        """
        _ = self.update(time_s)
        n_eclipse = sum(1 for sat in self.satellites if not sat.illuminated)
        return n_eclipse / self.n_satellites
```

```python
# core/energy_model.py
import numpy as np
from typing import Optional


class EnergyModel:
    """
    卫星能量模型
    基于MAAC-IILP [2] 方程(13)(14)(15) 和 DeepISL [3] 方程(7)(8)(9)
    """
    
    def __init__(self,
                 battery_max_kj: float = 100.0,
                 battery_min_kj: float = 10.0,
                 battery_init_fraction: float = 0.8,
                 solar_panel_area_m2: float = 0.5,
                 solar_efficiency: float = 0.28,
                 solar_irradiance_w_m2: float = 1353.0,
                 p_max_w: float = 4.0,
                 de_seconds: float = 30.0,
                 epsilon: float = 1e-3):
        
        self.C_max = battery_max_kj * 1000   # J
        self.C_min = battery_min_kj * 1000   # J
        self.C_init = battery_max_kj * battery_init_fraction * 1000  # J
        
        self.solar_area = solar_panel_area_m2
        self.solar_efficiency = solar_efficiency
        self.solar_irradiance = solar_irradiance_w_m2
        self.P_max = p_max_w
        self.delta_t = de_seconds
        self.epsilon = epsilon
        
    def compute_harvest_power(self,
                               illuminated: bool,
                               sun_angle_rad: float) -> float:
        """
        计算太阳能收集功率
        方程(13) from MAAC-IILP [2]:
        P_s = phi * eta * gamma' * A * sin(sigma)
        
        参数：
            illuminated: 光照指示符 phi in {0,1}
            sun_angle_rad: 太阳角度 sigma
        返回：
            P_harvest (W)
        """
        if not illuminated:
            return 0.0
        
        phi = 1.0
        P_harvest = (phi * self.solar_efficiency * 
                     self.solar_irradiance * 
                     self.solar_area * 
                     np.sin(sun_angle_rad))
        return max(0.0, P_harvest)
    
    def compute_max_power(self,
                           battery_level_j: float,
                           harvest_power_w: float,
                           illuminated: bool) -> float:
        """
        计算动态功率上限
        方程(9) from DeepISL [3]:
        
        Eclipse (phi=0): P_max_m,t = min(C_m(t)/delta(t), P_max)
        Illuminated (phi=1): P_max_m,t = min(P_harvest + C_m(t)/delta(t), P_max)
        
        参数：
            battery_level_j: 当前电量 C_m(t) (J)
            harvest_power_w: 太阳能收集功率 (W)
            illuminated: 光照状态
        返回：
            P_max_dynamic (W) - 本文关键创新点
        """
        battery_rate = battery_level_j / self.delta_t  # J/s = W
        
        if not illuminated:
            # Eclipse期间只能使用电池
            P_max_dynamic = min(battery_rate, self.P_max)
        else:
            # 光照期间可以使用太阳能+电池
            P_max_dynamic = min(harvest_power_w + battery_rate, self.P_max)
        
        return max(0.0, P_max_dynamic)
    
    def update_battery(self,
                        battery_level_j: float,
                        harvest_power_w: float,
                        total_consume_power_w: float) -> float:
        """
        更新电池状态
        方程(15) from MAAC-IILP [2] / 方程(12) from DeepISL [3]:
        
        delta_E = E_harvest - E_consume
        E_harvest = P_harvest * delta_t
        E_consume = sum_n P_mn * delta_t
        
        C(t+1) = min(C(t)+delta_E, C_max) if delta_E > 0
               = max(C(t)+delta_E, 0)      if delta_E <= 0
        
        参数：
            battery_level_j: 当前电量 (J)
            harvest_power_w: 收集功率 (W)
            total_consume_power_w: 总消耗功率 (W)
        返回：
            new_battery_level_j (J)
        """
        E_harvest = harvest_power_w * self.delta_t
        E_consume = total_consume_power_w * self.delta_t
        delta_E = E_harvest - E_consume
        
        new_level = battery_level_j + delta_E
        
        if delta_E > 0:
            new_level = min(new_level, self.C_max)
        else:
            new_level = max(new_level, 0.0)
        
        return new_level
    
    def compute_battery_penalty(self,
                                  battery_level_j: float,
                                  lambda_m: float) -> float:
        """
        计算电池惩罚项（本文效用函数核心）
        
        penalty = lambda_m / (C_m(t) - C_min + epsilon)
        
        当 C_m(t) → C_min 时，惩罚项趋于无穷
        这是本文的关键机制，提供软约束
        """
        denominator = battery_level_j - self.C_min + self.epsilon
        if denominator <= 0:
            return 1e10  # 数值保护
        return lambda_m / denominator
    
    def compute_esr(self, battery_trajectory: np.ndarray) -> float:
        """
        计算能量可持续率 ESR
        ESR = (1/M|T_e|) * sum_{m,t} 1[C_m(t) > C_min]
        
        参数：
            battery_trajectory: shape (M, T) - 电量轨迹 (J)
        返回：
            ESR in [0, 1]
        """
        above_min = battery_trajectory > self.C_min
        return float(np.mean(above_min))


class BatteryStateManager:
    """
    管理所有卫星的电池状态
    """
    
    def __init__(self, n_satellites: int, energy_model: EnergyModel):
        self.n_sats = n_satellites
        self.em = energy_model
        
        # 初始化电量
        self.battery_levels = np.full(
            n_satellites, energy_model.C_init
        )
        
        # 记录轨迹
        self.battery_history = [self.battery_levels.copy()]
        
    def step(self, 
             illuminated_flags: np.ndarray,
             sun_angles: np.ndarray,
             power_consumed: np.ndarray) -> np.ndarray:
        """
        执行一步电量更新
        
        参数：
            illuminated_flags: shape (M,) - 光照状态
            sun_angles: shape (M,) - 太阳角度 (rad)
            power_consumed: shape (M,) - 各卫星总消耗功率 (W)
        返回：
            new_battery_levels: shape (M,)
        """
        new_levels = np.zeros(self.n_sats)
        
        for m in range(self.n_sats):
            P_harvest = self.em.compute_harvest_power(
                illuminated_flags[m], sun_angles[m]
            )
            new_levels[m] = self.em.update_battery(
                self.battery_levels[m], P_harvest, power_consumed[m]
            )
        
        self.battery_levels = new_levels
        self.battery_history.append(new_levels.copy())
        
        return new_levels
    
    def get_dynamic_p_max(self,
                           illuminated_flags: np.ndarray,
                           sun_angles: np.ndarray) -> np.ndarray:
        """
        计算所有卫星的动态功率上限
        """
        p_max_dynamic = np.zeros(self.n_sats)
        
        for m in range(self.n_sats):
            P_harvest = self.em.compute_harvest_power(
                illuminated_flags[m], sun_angles[m]
            )
            p_max_dynamic[m] = self.em.compute_max_power(
                self.battery_levels[m], P_harvest, illuminated_flags[m]
            )
        
        return p_max_dynamic
    
    def reset(self):
        """重置电量状态"""
        self.battery_levels = np.full(
            self.n_sats, self.em.C_init
        )
        self.battery_history = [self.battery_levels.copy()]
```

```python
                 c_min_kbps: float = 10.0,
                 antenna_gain_dbw: float = 53.0):
        
        self.B = bandwidth_mhz * 1e6      # Hz
        self.f = carrier_freq_ghz * 1e9   # Hz
        self.tau = thermal_noise_k        # K
        self.P_max = p_max_w              # W
        self.C_min = c_min_kbps * 1e3    # bps
        self.c = 3e8                       # m/s
        self.k_B = 1.38e-23               # J/K
        
        # 天线增益（从dBW转换）
        # 简化：使用固定增益
        self.G_t = 10 ** (antenna_gain_dbw / 20)
        self.G_r = 10 ** (antenna_gain_dbw / 20)
        
    def compute_path_loss(self, distance_km: float) -> float:
        """
        计算自由空间路径损耗
        方程(1) from SATFLOW [1]:
        L_mn = (4*pi*d_mn*f/c)^2
        """
        d = distance_km * 1e3  # 转换为m
        L = (4 * np.pi * d * self.f / self.c) ** 2
        return L
    
    def compute_channel_capacity(self,
                                   distance_km: float,
                                   tx_power_w: float) -> float:
        """
        计算信道容量（Shannon公式）
        方程(2) from SATFLOW [1]:
        C_mn = B*log2(1 + p*G_t*G_r/(k_B*tau*B*L))
        """
        L = self.compute_path_loss(distance_km)
        
        if L <= 0 or tx_power_w <= 0:
            return 0.0
        
        SNR = tx_power_w * self.G_t * self.G_r / (self.k_B * self.tau * self.B * L)
        capacity = self.B * np.log2(1 + SNR)
        return capacity
    
    def compute_required_power(self,
                                 distance_km: float,
                                 data_rate_bps: float) -> float:
        """
        计算给定数据率所需的功率
        方程(3) from SATFLOW [1]:
        P_mn(C) = kappa_mn * (2^(C/B) - 1)
        
        kappa_mn = (k_B*tau*B)/(G_t*G_r) * L_mn
        """
        if data_rate_bps <= 0:
            return 0.0
        
        actual_rate = max(data_rate_bps, self.C_min)
        L = self.compute_path_loss(distance_km)
        kappa = (self.k_B * self.tau * self.B) / (self.G_t * self.G_r) * L
        P = kappa * (2 ** (actual_rate / self.B) - 1)
        return P
    
    def compute_kappa(self, distance_km: float) -> float:
        """计算kappa_mn参数"""
        L = self.compute_path_loss(distance_km)
        return (self.k_B * self.tau * self.B) / (self.G_t * self.G_r) * L
    
    def compute_energy_efficiency(self,
                                    data_rate_bps: float,
                                    tx_power_w: float) -> float:
        """
        计算能量效率 EE = R/P
        """
        if tx_power_w <= 0:
            return 0.0
        return data_rate_bps / tx_power_w
```

```python
# algorithms/proposed_pg.py
import numpy as np
from typing import List, Dict, Tuple, Optional
from scipy.optimize import minimize_scalar


class ProposedPG:
    """
    本文提出的势博弈算法（Energy-Aware SatFlow-L）
    
    实现论文[6]中：
    - 定义2（精确势博弈）
    - 定理1（精确势博弈性质）  
    - 定理2（O(1/k)收敛性）
    - Algorithm 1
    
    关键创新：
    1. 动态功率上限 P^max_{m,t}（方程7 of [6]）
    2. 电池感知惩罚项（效用函数11 of [6]）
    3. 势函数 Phi（方程13-14 of [6]）
    """
    
    def __init__(self,
                 n_satellites: int,
                 adjacency: np.ndarray,
                 comm_model,
                 energy_model,
                 lambda_energy: float = 0.1,
                 epsilon_penalty: float = 1e-3,
                 alpha: float = 2.0/3.0,
                 rho: float = 1e-10,
                 sigma: float = 1.9,
                 k_sg: int = 20,
                 k_max: int = 300,
                 tol: float = 1e-3,
                 de_seconds: float = 30.0):
        
        self.M = n_satellites
        self.A = adjacency            # 邻接矩阵
        self.comm = comm_model
        self.em = energy_model
        self.lambda_m = lambda_energy
        self.epsilon = epsilon_penalty
        self.alpha = alpha
        self.De = de_seconds
        self.rho = rho
        self.sigma = sigma
        self.k_sg = k_sg
        self.k_max = k_max
        self.tol = tol
        
        # 邻居列表
        self.neighbors = {
            m: list(np.where(adjacency[m] == 1)[0])
            for m in range(self.M)
        }
        
        # Linkage矩阵 L_m [1]方程(13)
        self.L = self._build_linkage_matrices()
        
    def _build_linkage_matrices(self) -> Dict[int, np.ndarray]:
        """
        构建链接矩阵 [L_m]_{M×4}
        基于SATFLOW [1] 方程(13)
        """
        L = {}
        for m in range(self.M):
            N_m = self.neighbors[m]
            n_neighbors = len(N_m)
            # 简化版本：每颗卫星最多4个邻居
            L_m = np.zeros((self.M, max(n_neighbors, 1)))
            L_m[m, :n_neighbors] = 1
            for j, n in enumerate(N_m):
                if j < L_m.shape[1]:
                    L_m[n, j] = -1
            L[m] = L_m
        return L
    
    def compute_battery_penalty_coefficient(self,
                                              battery_j: float,
                                              C_min_j: float) -> float:
        """
        计算电池惩罚系数：lambda / (C_m - C_min + epsilon)
        当C_m → C_min时趋于无穷（软约束机制）
        """
        denom = battery_j - C_min_j + self.epsilon
        if denom <= 0:
            return 1e10
        return self.lambda_m / denom
    
    def compute_utility(self,
                          Y: Dict[int, np.ndarray],
                          flows: list,
                          distances: np.ndarray,
                          battery_levels: np.ndarray,
                          lagrange_mult: np.ndarray,
                          C_min_j: float) -> float:
        """
        计算当前策略的总势函数值 Phi(R)
        方程(13-14) of [6]
        
        Phi(R) = sum_{m,n,t} R_mn 
               - alpha*De * sum_{m,n,t} P_mn(R_mn)
               - sum_{m,t} lambda_m/(C_m - C_min + eps) * sum_n P_mn
               - sum_{m,omega} <lambda^omega, L_m Y^omega_m>
        """
        total_throughput = 0.0
        total_energy_cost = 0.0
        total_battery_penalty = 0.0
        total_lagrangian = 0.0
        
        for m in range(self.M):
            battery_coeff = self.compute_battery_penalty_coefficient(
                battery_levels[m], C_min_j
            )
            
            total_power_m = 0.0
            
            for n in self.neighbors[m]:
                if Y[m] is None or len(Y[m]) == 0:
                    continue
                
                # 聚合所有流的数据率
                total_rate_mn = 0.0
                for omega, flow in enumerate(flows):
                    if omega < Y[m].shape[0] if Y[m].ndim > 1 else 1:
                        # Y[m]是分配矩阵，维度(n_flows, n_neighbors)
                        n_idx = self.neighbors[m].index(n) if n in self.neighbors[m] else -1
                        if n_idx >= 0 and Y[m].ndim > 1 and omega < Y[m].shape[0]:
                            total_rate_mn += max(0, Y[m][omega, n_idx])
                
                total_throughput += total_rate_mn
                
                # 计算功率
                d_mn = distances[m, n] if distances[m, n] > 0 else 1.0
                P_mn = self.comm.compute_required_power(d_mn, total_rate_mn)
                
                total_energy_cost += self.alpha * self.De * P_mn
                total_power_m += P_mn
                
                # Lagrangian项
                if lagrange_mult is not None:
                    total_lagrangian += np.sum(lagrange_mult * 0.0)  # 简化
            
            total_battery_penalty += battery_coeff * total_power_m
        
        phi = total_throughput - total_energy_cost - total_battery_penalty
        return phi
    
    def solve_local_subproblem(self,
                                 m: int,
                                 flows: list,
                                 distances: np.ndarray,
                                 battery_levels: np.ndarray,
                                 p_max_dynamic: np.ndarray,
                                 Y_current: Dict[int, np.ndarray],
                                 lagrange_mult: np.ndarray,
                                 C_min_j: float,
                                 b_vectors: Dict[int, np.ndarray]) -> np.ndarray:
        """
        求解卫星m的局部子问题（SatFlow-L第5行）
        
        min L_m(Y_hat_m, Y^(k), lambda^(k))
        
        使用投影次梯度法
        """
        N_m = self.neighbors[m]
        n_flows = len(flows)
        n_neighbors = len(N_m)
        
        if n_flows == 0 or n_neighbors == 0:
            return np.zeros((n_flows, n_neighbors))
        
        # 初始化
        Y_hat_m = Y_current[m].copy() if Y_current[m] is not None else \
                  np.zeros((n_flows, n_neighbors))
        
        battery_coeff = self.compute_battery_penalty_coefficient(
            battery_levels[m], C_min_j
        )
        
        # 投影次梯度迭代
        step_size_0 = 0.1
        
        for k_inner in range(self.k_sg):
            step_size = step_size_0 / (1 + k_inner * 0.1)
            
            # 计算梯度
            grad = np.zeros_like(Y_hat_m)
            
            for j, n in enumerate(N_m):
                d_mn = distances[m, n] if distances[m, n] > 0 else 1.0
                kappa_mn = self.comm.compute_kappa(d_mn)
                
                for omega in range(n_flows):
                    r = max(Y_hat_m[omega, j], 0)
                    
                    # dP/dR = kappa * ln2/B * 2^(R/B)
                    dP_dR = kappa_mn * np.log(2) / self.comm.B * \
                            2 ** (r / self.comm.B)
                    
                    # 目标函数梯度（最大化，取负梯度）
                    # dPhi/dR = 1 - (alpha*De + battery_coeff) * dP/dR
                    grad_phi = 1.0 - (self.alpha * self.De + battery_coeff) * dP_dR
                    
                    # Augmented Lagrangian梯度（简化）
                    grad_lag = self._compute_lagrangian_gradient(
                        m, n, j, omega, Y_hat_m, Y_current, 
                        lagrange_mult, b_vectors
                    )
                    
                    grad[omega, j] = grad_phi - grad_lag
            
            # 梯度上升（最大化势函数）
            Y_hat_m += step_size * grad
            
            # 投影到可行域 [0, R_max]
            for j, n in enumerate(N_m):
                d_mn = distances[m, n] if distances[m, n] > 0 else 1.0
                # 功率约束转换为速率约束
                R_max = self._power_to_rate(p_max_dynamic[m], d_mn)
                Y_hat_m[:, j] = np.clip(Y_hat_m[:, j], 0, R_max / n_flows)
        
        return Y_hat_m
    
    def _compute_lagrangian_gradient(self, m, n, j, omega,
                                       Y_hat_m, Y_current,
                                       lagrange_mult, b_vectors) -> float:
        """
        计算Augmented Lagrangian梯度项
        基于SATFLOW [1] 方程(17)
        """
        # 简化版本：计算flow conservation违反
        if lagrange_mult is None or b_vectors is None:
            return 0.0
        
        # 获取flow b向量
        b_omega = b_vectors.get(omega, {}).get(m, 0.0)
        
        # 计算残差
        residual = 0.0
        for i in range(self.M):
            if Y_current.get(i) is not None and Y_current[i].ndim > 1:
                # 简化：近似残差
                pass
        
        grad_lag = lagrange_mult[omega, m] + self.rho * residual
        return grad_lag
    
    def _power_to_rate(self, power_w: float, distance_km: float) -> float:
        """
        将功率约束转换为数据率约束（方程3的逆）
        R = B * log2(1 + P*G_t*G_r / (k_B*tau*B*L))
        """
        if power_w <= 0:
            return 0.0
        L = self.comm.compute_path_loss(distance_km)
        SNR = power_w * self.comm.G_t * self.comm.G_r / \
              (self.comm.k_B * self.comm.tau * self.comm.B * L)
        return self.comm.B * np.log2(1 + max(SNR, 0))
    
    def update_lagrange_multiplier(self,
                                     Y: Dict[int, np.ndarray],
                                     Y_hat: Dict[int, np.ndarray],
                                     lagrange_mult: np.ndarray,
                                     b_vectors: Dict,
                                     flows: list) -> np.ndarray:
        """
        更新Lagrange乘子
        方程(18) from SATFLOW [1]
        
        lambda^(k+1) = lambda^(k) + rho*sigma/q_m * 
                       (sum_i [L_i Y^hat_i]_m - [b]_m)
        """
        new_lambda = lagrange_mult.copy()
        n_flows = len(flows)
        
        for omega in range(n_flows):
            for m in range(self.M):
                # 计算flow conservation残差
                outflow = 0.0
                inflow = 0.0
                
                for n in self.neighbors[m]:
                    if Y_hat.get(m) is not None and Y_hat[m].ndim > 1:
                        j = self.neighbors[m].index(n) if n in self.neighbors[m] else -1
                        if j >= 0 and omega < Y_hat[m].shape[0]:
                            outflow += Y_hat[m][omega, j]
                
                b_m_omega = b_vectors.get(omega, {}).get(m, 0.0)
                residual = outflow - inflow - b_m_omega
                
                q_m = max(len(self.neighbors[m]), 1)
                new_lambda[omega, m] += self.rho * self.sigma / q_m * residual
        
        return new_lambda
    
    def run(self,
              flows: list,
              distances: np.ndarray,
              battery_levels: np.ndarray,
              p_max_dynamic: np.ndarray,
              C_min_j: float,
              b_vectors: Optional[Dict] = None) -> Dict:
        """
        运行Energy-Aware SatFlow-L (Algorithm 1 of [6])
        
        参数：
            flows: 流量列表
            distances: 距离矩阵 (M×M) km
            battery_levels: 当前电量 (M,) J
            p_max_dynamic: 动态功率上限 (M,) W
            C_min_j: 电池最低值 J
            b_vectors: flow conservation向量
        返回：
            result dict包含 Y, P, iterations, residuals
        """
        M = self.M
        n_flows = len(flows)
        
        if n_flows == 0:
            return {'Y': {}, 'P': np.zeros(M), 'iterations': 0, 
                    'residuals': [], 'converged': True}
        
        # 初始化 Y^(0) 和 lambda^(0)
        Y = {}
        for m in range(M):
            N_m = self.neighbors[m]
            n_neighbors = len(N_m)
            if n_neighbors > 0:
                Y[m] = np.zeros((n_flows, n_neighbors))
            else:
                Y[m] = np.zeros((n_flows, 0))
        
        lagrange_mult = np.zeros((n_flows, M))
        
        # b向量（flow conservation）
        if b_vectors is None:
            b_vectors = self._build_b_vectors(flows, M)
        
        residuals = []
        
        # 主迭代循环（Algorithm 1 Line 2-15）
        for k in range(self.k_max):
            Y_prev = {m: Y[m].copy() for m in range(M)}
            Y_hat = {}
            
            # 并行求解各卫星的局部子问题（Line 4-6）
            for m in range(M):
                Y_hat[m] = self.solve_local_subproblem(
                    m, flows, distances, battery_levels,
                    p_max_dynamic, Y, lagrange_mult, C_min_j, b_vectors
                )
            
            # 更新Lagrange乘子和原始变量（Line 7-10）
            lagrange_mult = self.update_lagrange_multiplier(
                Y, Y_hat, lagrange_mult, b_vectors, flows
            )
            
            for m in range(M):
                # Y^(k+1) = Y^(k) + sigma * (Y_hat^(k) - Y^(k))
                if Y[m].size > 0 and Y_hat[m].size > 0:
                    Y[m] = Y[m] + self.sigma * (Y_hat[m] - Y[m])
            
            # 计算收敛残差（Line 11）
            total_residual = 0.0
            for m in range(M):
                if Y[m].size > 0 and Y_prev[m].size > 0:
                    total_residual += np.linalg.norm(Y[m] - Y_prev[m])
            
            residuals.append(total_residual)
            
            # 检查收敛（Line 12-14）
            if total_residual <= self.tol:
                break
        
        # 计算功率分配
        P = self._compute_power_allocation(Y, distances, M)
        
        return {
            'Y': Y,
            'P': P,
            'iterations': k + 1,
            'residuals': residuals,
            'converged': total_residual <= self.tol if 'total_residual' in dir() else False
        }
    
    def _build_b_vectors(self, flows: list, M: int) -> Dict:
        """
        构建flow conservation的b向量
        基于SATFLOW [1] 方程(6)
        
        [b^omega]_a = +d_omega if a = src
                    = -d_omega if a = dst
                    = 0 otherwise
        """
        b_vectors = {}
        for omega, flow in enumerate(flows):
            b_omega = {}
            for m in range(M):
                if m == flow.src:
                    b_omega[m] = flow.rate_bps
                elif m == flow.dst:
                    b_omega[m] = -flow.rate_bps
                else:
                    b_omega[m] = 0.0
            b_vectors[omega] = b_omega
        return b_vectors
    
    def _compute_power_allocation(self,
                                    Y: Dict[int, np.ndarray],
                                    distances: np.ndarray,
                                    M: int) -> np.ndarray:
        """
        根据数据率分配计算功率消耗
        方程(3) from SATFLOW [1]
        """
        P = np.zeros(M)
        
        for m in range(M):
            if Y[m] is None or Y[m].size == 0:
                continue
            
            for j, n in enumerate(self.neighbors[m]):
                if j >= (Y[m].shape[1] if Y[m].ndim > 1 else 0):
                    continue
                
                total_rate = 0.0
                if Y[m].ndim > 1:
                    total_rate = np.sum(Y[m][:, j])
                
                d_mn = distances[m, n] if distances[m, n] > 0 else 1.0
                P_mn = self.comm.compute_required_power(d_mn, total_rate)
                P[m] += P_mn
        
        return P
    
    def compute_fvr(self, Y: Dict, flows: list, M: int) -> float:
        """
        计算流量违规率 FVR
        方程(23) from SATFLOW [1]
        """
        if not flows:
            return 0.0
        
        total_violation = 0.0
        n_flows = len(flows)
        
        for omega, flow in enumerate(flows):
            for m in range(M):
                outflow = 0.0
                inflow = 0.0
                
                for j, n in enumerate(self.neighbors[m]):
                    if Y.get(m) is not None and Y[m].ndim > 1 and j < Y[m].shape[1]:
                        if omega < Y[m].shape[0]:
                            outflow += max(0, Y[m][omega, j])
                
                for n in range(M):
                    if self.A[n, m] == 1:
                        j = self.neighbors[n].index(m) if m in self.neighbors[n] else -1
                        if j >= 0 and Y.get(n) is not None and Y[n].ndim > 1:
                            if j < Y[n].shape[1] and omega < Y[n].shape[0]:
                                inflow += max(0, Y[n][omega, j])
                
                b_m = 0.0
                if m == flow.src:
                    b_m = flow.rate_bps
                elif m == flow.dst:
                    b_m = -flow.rate_bps
                
                violation = abs(outflow - inflow - b_m)
                normalized = violation / (flow.rate_bps + 1e-10)
                total_violation += normalized
        
        fvr = total_violation / (2 * n_flows * M + 1e-10)
        return fvr
```

### 4.2 SATFLOW基准（静态P_max）

```python
# algorithms/satflow_l.py
import numpy as np
from typing import Dict, List, Optional
from algorithms.proposed_pg import ProposedPG


class SatFlowL(ProposedPG):
    """
    SATFLOW基准算法
    与本文方法的唯一区别：使用静态P_max而非动态P^max_{m,t}
    
    对应论文[1]的SatFlow-L，也是本文Proposition 1证明的对照
    """
    
    def __init__(self, p_max_static: float = 4.0, **kwargs):
        super().__init__(**kwargs)
        self.p_max_static = p_max_static
    
    def run(self,
              flows: list,
              distances: np.ndarray,
              battery_levels: np.ndarray,
              p_max_dynamic: np.ndarray,  # 忽略此参数
              C_min_j: float,
              b_vectors: Optional[Dict] = None) -> Dict:
        """
        使用静态P_max运行（约束(8)在[1]中）
        """
        # 关键差异：用静态P_max替换动态约束
        p_max_static_all = np.full(self.M, self.p_max_static)

        original_lambda = self.lambda_m
        self.lambda_m = 0.0  # 无电池感知
        
        result = super().run(
            flows, distances, battery_levels,
            p_max_static_all, C_min_j, b_vectors
        )

        return result
```

```python
# algorithms/maac_iilp.py
import numpy as np
import torch
import torch.nn as nn
import torch.optim as optim
from typing import Dict, List, Tuple
from collections import deque
import random


class AttentionCritic(nn.Module):
    """
    MAAC的注意力Critic网络
    基于MAAC-IILP [2] Section 4.2
    """
    
    def __init__(self, state_dim: int, action_dim: int, hidden_dim: int = 64,
                 n_agents: int = 10):
        super().__init__()
        
        # 嵌入层
        self.embedding = nn.Sequential(
            nn.Linear(state_dim + action_dim, hidden_dim),
            nn.ReLU()
        )
        
        # 注意力机制（参考[2] 方程(21)-(23)）
        self.W_k = nn.Linear(hidden_dim, hidden_dim, bias=False)
        self.W_q = nn.Linear(hidden_dim, hidden_dim, bias=False)
        self.W_v = nn.Linear(hidden_dim, hidden_dim, bias=False)
        
        # 输出层
        self.output = nn.Sequential(
            nn.Linear(hidden_dim * 2, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, 1)
        )
        
    def forward(self, states: torch.Tensor, actions: torch.Tensor,
                other_embeddings: torch.Tensor) -> torch.Tensor:
        # 计算当前agent嵌入
        sa = torch.cat([states, actions], dim=-1)
        e_i = self.embedding(sa)
        
        # 注意力机制
        q = self.W_q(e_i).unsqueeze(1)
        k = self.W_k(other_embeddings)
        v = self.W_v(other_embeddings)
        
        # 注意力权重
        attn_weights = torch.softmax(
            torch.bmm(q, k.transpose(1, 2)) / np.sqrt(e_i.shape[-1]), 
            dim=-1
        )
        
        # 聚合
        x_i = torch.bmm(attn_weights, v).squeeze(1)
        
        # Q值
        q_val = self.output(torch.cat([e_i, x_i], dim=-1))
        return q_val


class PolicyNetwork(nn.Module):
    """Actor网络"""
    
    def __init__(self, state_dim: int, action_dim: int, hidden_dim: int = 64):
        super().__init__()
        self.net = nn.Sequential(
            nn.Linear(state_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, action_dim),
            nn.Softmax(dim=-1)
        )
    
    def forward(self, state: torch.Tensor) -> torch.Tensor:
        return self.net(state)


class MAAcIilpBaseline:
    """
    MAAC-IILP基准实现（适配到mega-constellation规模）
    基于论文[2]
    
    状态空间扩展：增加C_m(t)和P^max_{m,t}（公平比较）
    """
    
    def __init__(self,
                 n_satellites: int,
                 adjacency: np.ndarray,
                 comm_model,
                 energy_model,
                 state_dim: int = 8,
                 action_dim: int = 5,
                 hidden_dim: int = 64,
                 n_episodes: int = 1000,
                 batch_size: int = 1024,
                 memory_size: int = 10000,
                 lr: float = 0.0025,
                 gamma: float = 0.99):
        
        self.M = n_satellites
        self.A = adjacency
        self.comm = comm_model
        self.em = energy_model
        self.n_episodes = n_episodes
        self.batch_size = batch_size
        self.gamma = gamma
        
        self.neighbors = {
            m: list(np.where(adjacency[m] == 1)[0])
            for m in range(n_satellites)
        }
        
        # 增强状态维度（扩充以包含电池状态）
        # S_i = {D_i, L_i, R_i, E_i, B_i, C_m(t), P^max_{m,t}}
        extended_state_dim = state_dim + 2  # +2 for C_m, P^max
        
        # 每个agent的网络
        self.policies = [
            PolicyNetwork(extended_state_dim, action_dim, hidden_dim)
            for _ in range(n_satellites)
        ]
        self.critics = [
            AttentionCritic(extended_state_dim, action_dim, hidden_dim)
            for _ in range(n_satellites)
        ]
        
        self.policy_optimizers = [
            optim.Adam(p.parameters(), lr=lr)
            for p in self.policies
        ]
        self.critic_optimizers = [
            optim.Adam(c.parameters(), lr=lr)
            for c in self.critics
        ]
        
        # 经验回放
        self.memory = deque(maxlen=memory_size)
        self.trained = False
        
    def get_state(self, m: int, distances: np.ndarray,
                   battery_level_j: float, p_max_dynamic_w: float) -> np.ndarray:
        """
        构建卫星m的状态向量
        基于[2] Section 4.1：
        S_i = {D_i, L_i, R_i, E_i, B_i}
        + 扩展：C_m(t), P^max_{m,t}（公平比较）
        """
        N_m = self.neighbors[m]
        
        # 归一化距离
        if N_m:
            distances_to_neighbors = [distances[m, n] / 1000.0 
                                       for n in N_m[:4]]
            while len(distances_to_neighbors) < 4:
                distances_to_neighbors.append(0.0)
        else:
            distances_to_neighbors = [0.0] * 4
        
        # 归一化电量
        battery_normalized = battery_level_j / self.em.C_max
        p_max_normalized = p_max_dynamic_w / self.em.P_max
        
        state = np.array(
            distances_to_neighbors[:4] + 
            [battery_normalized, p_max_normalized, 0.0, 0.0]
        )
        return state
    
    def select_action(self, state: np.ndarray, m: int,
                        epsilon: float = 0.1) -> int:
        """epsilon-greedy动作选择"""
        if np.random.random() < epsilon:
            return np.random.randint(0, len(self.neighbors[m]) + 1)
        
        with torch.no_grad():
            state_tensor = torch.FloatTensor(state).unsqueeze(0)
            probs = self.policies[m](state_tensor)
            return int(torch.argmax(probs).item())
    
    def train_step(self) -> float:
        """
        训练一步MAAC
        基于[2] Section 4.2
        """
        if len(self.memory) < self.batch_size:
            return 0.0
        
        batch = random.sample(self.memory, self.batch_size)
        total_loss = 0.0
        
        for m in range(self.M):
            states_m = torch.FloatTensor([b[0][m] for b in batch])
            actions_m = torch.LongTensor([b[1][m] for b in batch])
            rewards = torch.FloatTensor([b[2] for b in batch])
            next_states_m = torch.FloatTensor([b[3][m] for b in batch])
            
            # TD目标
            with torch.no_grad():
                next_actions = self.policies[m](next_states_m)
                next_q = self.critics[m](
                    next_states_m, next_actions,
                    next_states_m.unsqueeze(1).repeat(1, 1, 1)
                )
                targets = rewards.unsqueeze(1) + self.gamma * next_q
            
            # Critic更新
            actions_onehot = torch.zeros(self.batch_size, 
                                          self.policies[m].net[-2].out_features)
            actions_onehot.scatter_(1, actions_m.unsqueeze(1), 1.0)
            
            current_q = self.critics[m](
                states_m, actions_onehot,
                states_m.unsqueeze(1).repeat(1, 1, 1)
            )
            
            critic_loss = nn.MSELoss()(current_q, targets)
            self.critic_optimizers[m].zero_grad()
            critic_loss.backward()
            self.critic_optimizers[m].step()
            
            # Actor更新
            policy_actions = self.policies[m](states_m)
            actor_loss = -self.critics[m](
                states_m, policy_actions,
                states_m.unsqueeze(1).repeat(1, 1, 1)
            ).mean()
            
            self.policy_optimizers[m].zero_grad()
            actor_loss.backward()
            self.policy_optimizers[m].step()
            
            total_loss += critic_loss.item() + actor_loss.item()
        
        return total_loss / self.M
    
    def run(self,
              flows: list,
              distances: np.ndarray,
              battery_levels: np.ndarray,
              p_max_dynamic: np.ndarray,
              C_min_j: float,
              b_vectors=None) -> Dict:
        """
        运行MAAC-IILP策略
        
        如果未训练，先进行训练
        返回与ProposedPG相同格式的结果
        """
        M = self.M
        n_flows = len(flows)
        
        if n_flows == 0:
            return {
                'Y': {m: np.zeros((0, len(self.neighbors[m]))) for m in range(M)},
                'P': np.zeros(M),
                'iterations': 0,
                'residuals': [],
                'converged': True
            }
        
        # 获取当前状态并执行策略
        Y = {}
        for m in range(M):
            N_m = self.neighbors[m]
            n_neighbors = len(N_m)
            
            state = self.get_state(m, distances, battery_levels[m], 
                                    p_max_dynamic[m])
            
            if not self.trained:
                # 使用启发式初始化（等分带宽）
                Y[m] = np.zeros((n_flows, n_neighbors))
                if n_neighbors > 0 and n_flows > 0:
                    for j, n in enumerate(N_m):
                        d_mn = distances[m, n] if distances[m, n] > 0 else 1.0
                        # 在功率约束内平均分配
                        R_max = self._power_to_rate(p_max_dynamic[m], d_mn)
                        Y[m][:, j] = R_max / (n_flows * n_neighbors)
            else:
                # 使用学习到的策略
                action = self.select_action(state, m, epsilon=0.0)
                Y[m] = self._action_to_allocation(
                    action, m, n_flows, distances, p_max_dynamic[m]
                )
        
        P = self._compute_power(Y, distances, M)
        
        return {
            'Y': Y,
            'P': P,
            'iterations': 1,
            'residuals': [0.0],
            'converged': True
        }
    
    def _power_to_rate(self, power_w: float, distance_km: float) -> float:
        """功率到速率转换"""
        if power_w <= 0:
            return 0.0
        L = self.comm.compute_path_loss(distance_km)
        SNR = power_w * self.comm.G_t * self.comm.G_r / \
              (self.comm.k_B * self.comm.tau * self.comm.B * L)
        return self.comm.B * np.log2(1 + max(SNR, 0))
    
    def _action_to_allocation(self, action: int, m: int, n_flows: int,
                               distances: np.ndarray, 
                               p_max_w: float) -> np.ndarray:
        """将离散动作映射到连续分配"""
        N_m = self.neighbors[m]
        n_neighbors = len(N_m)
        Y = np.zeros((n_flows, n_neighbors))
        
        if action < n_neighbors and n_neighbors > 0:
            # 将资源分配给选定的邻居方向
            n = N_m[action]
            d_mn = distances[m, n] if distances[m, n] > 0 else 1.0
            R_max = self._power_to_rate(p_max_w, d_mn)
            Y[:, action] = R_max / n_flows
        
        return Y
    
    def _compute_power(self, Y: Dict, distances: np.ndarray, M: int) -> np.ndarray:
        """计算功率消耗"""
        P = np.zeros(M)
        for m in range(M):
            if Y[m] is None or Y[m].size == 0:
                continue
            for j, n in enumerate(self.neighbors[m]):
                if j >= (Y[m].shape[1] if Y[m].ndim > 1 else 0):
                    continue
                if Y[m].ndim > 1:
                    total_rate = np.sum(Y[m][:, j])
                    d_mn = distances[m, n] if distances[m, n] > 0 else 1.0
                    P[m] += self.comm.compute_required_power(d_mn, total_rate)
        return P
```

```python
# algorithms/deepisl.py
import numpy as np
import torch
import torch.nn as nn
import torch.optim as optim
from typing import Dict, List, Optional


class PdqnNetwork(nn.Module):
    """
    P-DQN网络（参数化DQN，处理混合动作空间）
    基于DeepISL [3] Section IV-B
    """
    
    def __init__(self, state_dim: int, n_discrete: int, 
                 hidden_dim: int = 64):
        super().__init__()
        
        # Q网络
        self.q_net = nn.Sequential(
            nn.Linear(state_dim + 1, hidden_dim),  # +1 for continuous action
            nn.ReLU(),
            nn.Linear(hidden_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, n_discrete)
        )
        
        # 策略网络（连续功率动作）
        self.policy_net = nn.Sequential(
            nn.Linear(state_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, n_discrete),
            nn.Sigmoid()  # 功率在[0,1]范围内
        )
    
    def q_value(self, state: torch.Tensor, 
                power: torch.Tensor) -> torch.Tensor:
        """计算Q值（方程22 of [3]）"""
        sa = torch.cat([state, power.unsqueeze(-1)], dim=-1)
        return self.q_net(sa)
    
    def policy(self, state: torch.Tensor) -> torch.Tensor:
        """输出连续功率参数"""
        return self.policy_net(state)


class DeepISLBaseline:
    """
    DeepISL基准实现
    基于论文[3]
    
    适配扩展：
    - 连续功率动作被裁剪至 [0, P^max_{m,t}]（公平比较）
    """
    
    def __init__(self,
                 n_satellites: int,
                 adjacency: np.ndarray,
                 comm_model,
                 energy_model,
                 hidden_dim: int = 64,
                 lr_q: float = 0.0095,
                 lr_pi: float = 0.0095,
                 gamma: float = 0.95,
                 epsilon: float = 0.8,
                 memory_size: int = 10000,
                 batch_size: int = 1024,
                 satisfaction_factor: float = 0.9):
        
        self.M = n_satellites
        self.A = adjacency
        self.comm = comm_model
        self.em = energy_model
        self.gamma = gamma
        self.epsilon = epsilon
        self.batch_size = batch_size
        self.lambda_sf = satisfaction_factor
        
        self.neighbors = {
            m: list(np.where(adjacency[m] == 1)[0])
            for m in range(n_satellites)
        }
        
        # 状态维度：{D_i, C_i, omega_i}（参考[3] Section IV-A）
        # + C_m(t), P^max_{m,t} 扩展
        state_dim = 7  # 4距离 + 1电量 + 1数据量 + 1P_max
        
        self.networks = []
        self.q_optimizers = []
        self.pi_optimizers = []
        
        for m in range(n_satellites):
            n_discrete = max(len(self.neighbors[m]), 1)
            net = PdqnNetwork(state_dim, n_discrete, hidden_dim)
            self.networks.append(net)
            self.q_optimizers.append(optim.Adam(net.q_net.parameters(), lr=lr_q))
            self.pi_optimizers.append(optim.Adam(net.policy_net.parameters(), lr=lr_pi))
        
        from collections import deque
        self.memory = deque(maxlen=memory_size)
        self.trained = False
        
    def get_state(self, m: int, distances: np.ndarray,
                   battery_j: float, p_max_w: float,
                   data_amount: float = 1.0) -> np.ndarray:
        """
        构建状态向量
        S_i = {D_i, C_i, omega_i} + 扩展
        """
        N_m = self.neighbors[m]
        
        dist_feats = []
        for n in N_m[:4]:
            dist_feats.append(distances[m, n] / 1000.0)
        while len(dist_feats) < 4:
            dist_feats.append(0.0)
        
        return np.array(
            dist_feats + 
            [battery_j / self.em.C_max,
             data_amount,
             p_max_w / self.em.P_max]
        )
    
    def select_action(self, state: np.ndarray, m: int,
                        p_max_w: float) -> Tuple:
        """
        选择混合动作 (v_i, p_vi)
        参考[3] Section IV-A的参数化动作空间
        """
        N_m = self.neighbors[m]
        
        if len(N_m) == 0:
            return 0, 0.0
        
        state_tensor = torch.FloatTensor(state).unsqueeze(0)
        
        # 获取连续功率参数
        with torch.no_grad():
            power_params = self.networks[m].policy(state_tensor)[0]  # (n_discrete,)
            power_params = power_params.numpy()
        
        if np.random.random() < self.epsilon:
            # 探索
            vi = np.random.randint(0, len(N_m))
            pvi = np.random.uniform(0, p_max_w)
        else:
            # 利用（方程8 of [3]）
            vi = int(np.argmax([
                self.networks[m].q_value(
                    state_tensor,
                    torch.FloatTensor([power_params[j]])
                )[0, j].item()
                for j in range(len(N_m))
            ]))
            pvi = float(power_params[vi]) * p_max_w
        
        # 裁剪到动态功率上限（公平比较扩展）
        pvi = np.clip(pvi, 0, p_max_w)
        
        return vi, pvi
    
    def run(self,
              flows: list,
              distances: np.ndarray,
              battery_levels: np.ndarray,
              p_max_dynamic: np.ndarray,
              C_min_j: float,
              b_vectors=None) -> Dict:
        """运行DeepISL策略"""
        M = self.M
        n_flows = len(flows)
        
        if n_flows == 0:
            return {
                'Y': {m: np.zeros((0, len(self.neighbors[m]))) for m in range(M)},
                'P': np.zeros(M),
                'iterations': 0,
                'residuals': [],
                'converged': True
            }
        
        Y = {}
        P = np.zeros(M)
        
        for m in range(M):
            N_m = self.neighbors[m]
            n_neighbors = len(N_m)
            Y[m] = np.zeros((n_flows, n_neighbors))
            
            if n_neighbors == 0:
                continue
            
            state = self.get_state(
                m, distances, battery_levels[m], p_max_dynamic[m]
            )
            
            # 选择动作
            vi, pvi = self.select_action(state, m, p_max_dynamic[m])
            
            if vi < n_neighbors:
                n = N_m[vi]
                d_mn = distances[m, n] if distances[m, n] > 0 else 1.0
                
                # 计算数据率（方程6 of [3]，带满足因子约束）
                capacity = self.comm.compute_channel_capacity(d_mn, pvi)
                
                # 应用满足因子约束（方程14 of [3]）
                # lambda * omega/delta <= R <= omega/delta
                for omega in range(n_flows):
                    omega_rate = flows[omega].rate_bps
                    min_rate = self.lambda_sf * omega_rate
                    max_rate_flow = omega_rate
                    
                    assigned_rate = np.clip(
                        capacity / n_flows, min_rate, max_rate_flow
                    )
                    Y[m][omega, vi] = assigned_rate
                
                P[m] = pvi
        
        return {
            'Y': Y,
            'P': P,
            'iterations': 1,
            'residuals': [0.0],
            'converged': True
        }
```

```python
# algorithms/smfg_adapted.py
import numpy as np
from typing import Dict, List, Optional
from scipy.optimize import minimize


class SmfgAdaptedBaseline:
    """
    SMFG-adapted基准
    
    改编自SMFG [5]：
    - 移除Stackelberg层（卫星定价）
    - 用对等卫星替换地面用户
    - 使用G-prox PDHG求解功率控制
    
    基于[5] Section V-A的G-prox PDHG算法
    """
    
    def __init__(self,
                 n_satellites: int,
                 adjacency: np.ndarray,
                 comm_model,
                 energy_model,
                 n_iter: int = 100,
                 step_size: float = 0.1,
                 epsilon: float = 1e-3):
        
        self.M = n_satellites
        self.A = adjacency
        self.comm = comm_model
        self.em = energy_model
        self.n_iter = n_iter
        self.step_size = step_size
        self.epsilon = epsilon
        
        self.neighbors = {
            m: list(np.where(adjacency[m] == 1)[0])
            for m in range(n_satellites)
        }
        
    def compute_mean_field(self, battery_levels: np.ndarray) -> float:
        """
        计算平均场（电量分布均值）
        对应[5]方程(15)中的mean field term
        """
        return float(np.mean(battery_levels / self.em.C_max))
    
    def utility(self, power_w: float, m: int,
                  distances: np.ndarray,
                  battery_j: float,
                  mean_field: float,
                  kappa: float = 1.0) -> float:
        """
        卫星效用函数（G-prox PDHG的目标）
        改编自[5]方程(9)，针对ISL场景
        
        对比[5]中地面用户的负功耗和费用，
        这里改为：-power - battery_penalty + throughput
        """
        N_m = self.neighbors[m]
        if not N_m:
            return 0.0
        
        # 能量成本
        energy_cost = power_w
        
        # 电池惩罚（类似[5]中的数据状态惩罚）
        battery_penalty = kappa * abs(battery_j / self.em.C_max - mean_field)
        
        # 吞吐量收益
        avg_dist = np.mean([distances[m, n] for n in N_m if distances[m, n] > 0])
        if avg_dist > 0:
            throughput = self.comm.compute_channel_capacity(avg_dist, power_w)
        else:
            throughput = 0.0
        
        return throughput - energy_cost - battery_penalty
    
    def run(self,
              flows: list,
              distances: np.ndarray,
              battery_levels: np.ndarray,
              p_max_dynamic: np.ndarray,
              C_min_j: float,
              b_vectors=None) -> Dict:
        """
        使用G-prox PDHG求解功率分配
        
        基于[5] Algorithm 1（地面用户版本）
        适配到卫星间ISL竞争
        """
        M = self.M
        n_flows = len(flows)
        
        if n_flows == 0:
            return {
                'Y': {m: np.zeros((0, len(self.neighbors[m]))) for m in range(M)},
                'P': np.zeros(M),
                'iterations': 0,
                'residuals': [],
                'converged': True
            }
        
        # 计算平均场
        mean_field = self.compute_mean_field(battery_levels)
        
        # 初始化功率分配
        powers = np.array([p_max_dynamic[m] * 0.5 for m in range(M)])
        
        # G-prox PDHG迭代（[5] Algorithm 1的卫星版本）
        for k in range(self.n_iter):
            new_powers = powers.copy()
            
            for m in range(M):
                # 最优功率（通过梯度上升）
                def neg_utility(p):
                    return -self.utility(p[0], m, distances, 
                                          battery_levels[m], mean_field)
                
                from scipy.optimize import minimize_scalar
                result = minimize_scalar(
                    lambda p: -self.utility(p, m, distances,
                                              battery_levels[m], mean_field),
                    bounds=(0, p_max_dynamic[m]),
                    method='bounded'
                )
                new_powers[m] = result.x
            
            # 收敛检查
            if np.linalg.norm(new_powers - powers) < self.epsilon:
                break
            
            powers = new_powers
        
        # 将功率分配转换为数据率分配
        Y = {}
        for m in range(M):
            N_m = self.neighbors[m]
            n_neighbors = len(N_m)
            Y[m] = np.zeros((n_flows, n_neighbors))
            
            if n_neighbors == 0 or powers[m] <= 0:
                continue
            
            # 等分功率给各邻居和流
            power_per_link = powers[m] / n_neighbors
            
            for j, n in enumerate(N_m):
                d_mn = distances[m, n] if distances[m, n] > 0 else 1.0
                capacity = self.comm.compute_channel_capacity(d_mn, power_per_link)
                
                for omega in range(n_flows):
                    Y[m][omega, j] = capacity / n_flows
        
        return {
            'Y': Y,
            'P': powers,
            'iterations': k + 1,
            'residuals': [0.0],
            'converged': True
        }
```

```python
# algorithms/proposed_mfg.py
import numpy as np
from typing import Dict, Optional
from algorithms.proposed_pg import ProposedPG


class ProposedMFG(ProposedPG):
    """
    本文MFG扩展方法
    
    实现论文[6]中：
    - 方程(25)：经验测度
    - HJB-FPK系统（方程26）
    - 定理3：O(M^{-1/4})收敛性
    - Lemma 1和2
    
    关键：当M→∞时，将有限玩家博弈约化为MFG
    平均场变量：电池状态分布 mu(C_B, t)
    """
    
    def __init__(self, mfg_alpha: float = 1.0, **kwargs):
        super().__init__(**kwargs)
        self.mfg_alpha = mfg_alpha  # Wasserstein耦合系数 kappa
        
        # 平均场状态（电池分布的统计量）
        self.mean_field_mean = None
        self.mean_field_std = None
        
    def compute_empirical_measure(self, 
                                    battery_levels: np.ndarray) -> Dict:
        """
        计算经验测度 mu^M(C_B, t)
        方程(25) of [6]:
        mu^M(C_B, t) = (1/M) * sum_m delta_{C_m(t)}
        
        返回统计量（用于MFG近似）
        """
        normalized_levels = battery_levels / self.em.C_max
        return {
            'mean': float(np.mean(normalized_levels)),
            'std': float(np.std(normalized_levels)),
            'histogram': np.histogram(normalized_levels, bins=10, range=(0, 1))[0] / self.M
        }
    
    def compute_wasserstein_coupling(self,
                                       battery_j: float,
                                       mean_field: Dict) -> float:
        """
        计算Wasserstein距离耦合项 d(C_m, mu)
        
        在HJB方程(26a)中：
        -kappa * d(C_m, mu(C_B, t))
        
        使用简化的1D Wasserstein近似
        """
        normalized = battery_j / self.em.C_max
        mu_mean = mean_field['mean']
        mu_std = mean_field['std']
        
        # 一维Wasserstein距离近似
        # W_1(delta_x, mu) ≈ |x - mu_mean| (当mu接近正态时)
        wasserstein_dist = abs(normalized - mu_mean)
        
        return float(self.mfg_alpha * wasserstein_dist)
    
    def compute_mfg_adjusted_utility(self,
                                       rate_bps: float,
                                       power_w: float,
                                       battery_j: float,
                                       mean_field: Dict,
                                       C_min_j: float) -> float:
        """
        计算MFG调整后的效用
        
        基于HJB方程(26a) of [6]:
        max_R [R - (alpha*De + lambda/(C_m-C_min+eps))*P(R) 
               - kappa*d(C_m, mu)]
        
        相比势博弈版本增加了均场耦合项
        """
        # 吞吐量收益
        throughput = rate_bps
        
        # 能量代价
        energy_cost = (self.alpha * self.De + 
                        self.compute_battery_penalty_coefficient(battery_j, C_min_j)) * power_w
        
        # Wasserstein耦合项（MFG特有）
        wasserstein_term = self.compute_wasserstein_coupling(battery_j, mean_field)
        
        return throughput - energy_cost - wasserstein_term
    
    def run(self,
              flows: list,
              distances: np.ndarray,
              battery_levels: np.ndarray,
              p_max_dynamic: np.ndarray,
              C_min_j: float,
              b_vectors: Optional[Dict] = None) -> Dict:
        """
        运行MFG扩展算法
        
        相比ProposedPG，在效用函数中加入均场耦合
        实现定理3中的近最优集体策略
        """
        # 计算平均场
        mean_field = self.compute_empirical_measure(battery_levels)
        self.mean_field_mean = mean_field['mean']
        self.mean_field_std = mean_field['std']
        
        # 存储平均场以供局部子问题使用
        self._current_mean_field = mean_field
        self._current_C_min_j = C_min_j
        
        # 调用父类的run，但使用MFG调整后的效用
        result = super().run(
            flows, distances, battery_levels,
            p_max_dynamic, C_min_j, b_vectors
        )
        
        result['mean_field'] = mean_field
        return result
    
    def solve_local_subproblem(self, m, flows, distances, battery_levels,
                                 p_max_dynamic, Y_current, lagrange_mult,
                                 C_min_j, b_vectors):
        """
        覆盖父类方法，在梯度计算中加入MFG耦合项
        """
        N_m = self.neighbors[m]
        n_flows = len(flows)
        n_neighbors = len(N_m)
        
        if n_flows == 0 or n_neighbors == 0:
            return np.zeros((n_flows, n_neighbors))
        
        Y_hat_m = Y_current[m].copy() if Y_current[m] is not None else \
                  np.zeros((n_flows, n_neighbors))
        
        battery_coeff = self.compute_battery_penalty_coefficient(
            battery_levels[m], C_min_j
        )
        
        # 获取当前平均场
        mean_field = getattr(self, '_current_mean_field', 
                              {'mean': 0.5, 'std': 0.1})
        
        # MFG额外梯度：Wasserstein耦合的梯度
        wasserstein_grad = self.compute_wasserstein_coupling(
            battery_levels[m], mean_field
        )
        
        step_size_0 = 0.1
        
        for k_inner in range(self.k_sg):
            step_size = step_size_0 / (1 + k_inner * 0.1)
            grad = np.zeros_like(Y_hat_m)
            
            for j, n in enumerate(N_m):
                d_mn = distances[m, n] if distances[m, n] > 0 else 1.0
                kappa_mn = self.comm.compute_kappa(d_mn)
                
                for omega in range(n_flows):
                    r = max(Y_hat_m[omega, j], 0)
                    dP_dR = kappa_mn * np.log(2) / self.comm.B * \
                            2 ** (r / self.comm.B)
                    
                    # MFG调整梯度（增加Wasserstein项，作为常数近似）
                    grad_phi = (1.0 - (self.alpha * self.De + battery_coeff) * dP_dR
                                - wasserstein_grad * 0.01)  # 小的均场影响系数
                    
                    grad_lag = self._compute_lagrangian_gradient(
                        m, n, j, omega, Y_hat_m, Y_current,
                        lagrange_mult, b_vectors
                    )
                    
                    grad[omega, j] = grad_phi - grad_lag
            
            Y_hat_m += step_size * grad
            
            for j, n in enumerate(N_m):
                d_mn = distances[m, n] if distances[m, n] > 0 else 1.0
                R_max = self._power_to_rate(p_max_dynamic[m], d_mn)
                Y_hat_m[:, j] = np.clip(Y_hat_m[:, j], 0, R_max / n_flows)
        
        return Y_hat_m
    
    def _power_to_rate(self, power_w: float, distance_km: float) -> float:
        """功率到速率转换"""
        if power_w <= 0:
            return 0.0
        L = self.comm.compute_path_loss(distance_km)
        SNR = power_w * self.comm.G_t * self.comm.G_r / \
              (self.comm.k_B * self.comm.tau * self.comm.B * L)
        return self.comm.B * np.log2(1 + max(SNR, 0))
```

```python
# utils/metrics.py
import numpy as np
from typing import Dict, List


class MetricsCalculator:
    """
    计算论文中的所有评估指标
    """
    
    def __init__(self, energy_model, comm_model, de_seconds: float = 30.0):
        self.em = energy_model
        self.comm = comm_model
        self.De = de_seconds
    
    def compute_esr(self, battery_trajectory: np.ndarray) -> float:
        """
        能量可持续率 ESR（本文核心指标）
        方程(37) of [6]:
        ESR = (1/M|T_e|) * sum_{m,t} 1[C_m(t) > C_min]
        
        参数：
            battery_trajectory: shape (T, M) - 电量轨迹 (J)
        """
        T, M = battery_trajectory.shape
        above_min = battery_trajectory > self.em.C_min
        return float(np.mean(above_min))
    
    def compute_fvr(self, Y: Dict, flows: list, 
                      adjacency: np.ndarray, M: int) -> float:
        """
        流量违规率 FVR
        方程(23) from SATFLOW [1]
        """
        if not flows:
            return 0.0
        
        neighbors = {m: list(np.where(adjacency[m] == 1)[0]) 
                     for m in range(M)}
        
        total_violation = 0.0
        n_flows = len(flows)
        
        for omega, flow in enumerate(flows):
            for m in range(M):
                outflow = 0.0
                inflow = 0.0
                
                # 出流
                if Y.get(m) is not None and Y[m].ndim > 1:
                    for j, n in enumerate(neighbors[m]):
                        if j < Y[m].shape[1] and omega < Y[m].shape[0]:
                            outflow += max(0, Y[m][omega, j])
                
                # 入流
                for n in range(M):
                    if adjacency[n, m] == 1:
                        if Y.get(n) is not None and Y[n].ndim > 1:
                            j_idx = neighbors[n].index(m) if m in neighbors[n] else -1
                            if j_idx >= 0 and j_idx < Y[n].shape[1] and omega < Y[n].shape[0]:
                                inflow += max(0, Y[n][omega, j_idx])
                
                # b向量
                b_m = 0.0
                if m == flow.src:
                    b_m = flow.rate_bps
                elif m == flow.dst:
                    b_m = -flow.rate_bps
                
                violation = abs(outflow - inflow - b_m)
                total_violation += violation / (flow.rate_bps + 1e-10)
        
        return total_violation / (2 * n_flows * M + 1e-10)
    
    def compute_energy_efficiency(self, Y: Dict, 
                                    distances: np.ndarray,
                                    adjacency: np.ndarray,
                                    M: int,
                                    flows: list) -> float:
        """
        能量效率 EE = 总吞吐量 / 总能耗
        """
        neighbors = {m: list(np.where(adjacency[m] == 1)[0]) 
                     for m in range(M)}
        
        total_throughput = 0.0
        total_energy = 0.0
        
        for m in range(M):
            if Y.get(m) is None or Y[m].size == 0:
                continue
            
            for j, n in enumerate(neighbors[m]):
                if j >= (Y[m].shape[1] if Y[m].ndim > 1 else 0):
                    continue
                
                if Y[m].ndim > 1:
                    total_rate = np.sum(Y[m][:, j])
                    total_throughput += total_rate
                    
                    d_mn = distances[m, n] if distances[m, n] > 0 else 1.0
                    power = self.comm.compute_required_power(d_mn, total_rate)
                    total_energy += power * self.De
        
        if total_energy <= 0:
            return 0.0
        
        # 转换为Mbit/J
        return (total_throughput * self.De / 1e6) / total_energy
    
    def compute_normalized_cost(self, Y: Dict,
                                   distances: np.ndarray,
                                   adjacency: np.ndarray,
                                   M: int,
                                   flows: list,
                                   alpha: float = 2/3,
                                   beta: float = 1/3) -> float:
        """
        归一化总成本（包含能量消耗和ISL切换成本）
        基于SATFLOW [1] 方程(7)
        """
        neighbors = {m: list(np.where(adjacency[m] == 1)[0]) 
                     for m in range(M)}
        
        total_energy = 0.0
        switching_cost = 0.0  # 简化：假设已知
        
        for m in range(M):
            if Y.get(m) is None:
                continue
            for j, n in enumerate(neighbors[m]):
                if j >= (Y[m].shape[1] if Y[m].ndim > 1 else 0):
                    continue
                if Y[m].ndim > 1:
                    total_rate = np.sum(Y[m][:, j])
                    d_mn = distances[m, n] if distances[m, n] > 0 else 1.0
                    power = self.comm.compute_required_power(d_mn, total_rate)
                    total_energy += power * self.De
        
        # 归一化
        cost = alpha * total_energy / (M * 1e3) + beta * switching_cost
        return cost
    
    def compute_all_metrics(self, 
                              Y: Dict,
                              distances: np.ndarray,
                              battery_trajectory: np.ndarray,
                              adjacency: np.ndarray,
                              M: int,
                              flows: list) -> Dict:
        """计算所有指标"""
        return {
            'ESR': self.compute_esr(battery_trajectory),
            'FVR': self.compute_fvr(Y, flows, adjacency, M),
            'EE_mbit_per_J': self.compute_energy_efficiency(
                Y, distances, adjacency, M, flows
            ),
            'Cost': self.compute_normalized_cost(
                Y, distances, adjacency, M, flows
            )
        }
```

```python
def compute_confidence_interval(data: list, confidence: float = 0.95):
    """计算置信区间"""
    n = len(data)
    mean = np.mean(data)
    std = np.std(data, ddof=1)
    t_val = stats.t.ppf((1 + confidence) / 2, df=n-1)
    margin = t_val * std / np.sqrt(n)
    return mean, margin
```

```python
            res_vals = np.log(np.maximum(residuals, 1e-20))
            slope = np.polyfit(k_vals[-50:], res_vals[-50:], 1)[0]
            print(f"lambda={lam}: 收敛率斜率 ≈ {slope:.3f} (理论 -0.5)")


if __name__ == '__main__':
    run_fig4()
```

```python
# experiments/run_fig5.py
"""
生成Figure 5：运行时间 vs 卫星数量
验证Proposition 2：O(n_t)线性复杂度
"""
import numpy as np
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt
import time
import sys
import os

from config import CFG
from core.constellation import Constellation
from core.energy_model import EnergyModel, BatteryStateManager
from core.communication import CommunicationModel
from core.traffic import TrafficGenerator
from algorithms.proposed_pg import ProposedPG


def run_fig5():
    """测量不同星座规模下的运行时间"""
    
    # 测试三个星座规模
    test_constellations = {
        'Starlink Shell A\n(172 sats)': 'starlink_a',   # 4*43=172
        'Starlink Shell B\n(348 sats)': 'starlink_b',   # 6*58=348
        'Kuiper\n(784 sats)': 'kuiper'                   # 28*28=784
    }
    
    runtimes = {}
    n_satellites_list = []
    
    for label, const_name in test_constellations.items():
        const_config = CFG.CONSTELLATIONS[const_name]
        constellation = Constellation(const_config)
        M = constellation.n_satellites
        n_satellites_list.append(M)
        
        em = EnergyModel(
            battery_max_kj=CFG.BATTERY_MAX_KJ,
            battery_min_kj=CFG.BATTERY_MIN_KJ,
            battery_init_fraction=0.8,
            solar_panel_area_m2=CFG.SOLAR_PANEL_AREA_M2,
            solar_efficiency=CFG.SOLAR_EFFICIENCY,
            solar_irradiance_w_m2=CFG.SOLAR_IRRADIANCE,
            p_max_w=CFG.P_MAX_W,
            de_seconds=CFG.DE_SECONDS
        )
        
        comm = CommunicationModel(
            bandwidth_mhz=CFG.BANDWIDTH_MHZ,
            carrier_freq_ghz=CFG.CARRIER_FREQ_GHZ,
            thermal_noise_k=CFG.THERMAL_NOISE_K,
            p_max_w=CFG.P_MAX_W,
            c_min_kbps=CFG.C_MIN_KBPS
        )
        
        traffic_gen = TrafficGenerator(M, CFG)
        n_flows = traffic_gen.sample_n_flows(const_name, 'medium', seed=42)
        flows = traffic_gen.generate_flows(n_flows, const_name, seed=42)
        
        algorithm = ProposedPG(
            n_satellites=M,
            adjacency=constellation.adjacency,
            comm_model=comm,
            energy_model=em,
            lambda_energy=CFG.LAMBDA_ENERGY,
            epsilon_penalty=CFG.EPSILON_PENALTY,
            alpha=CFG.ALPHA_WEIGHT,
            rho=CFG.RHO,
            sigma=CFG.SIGMA,
            k_sg=CFG.K_SG,
            k_max=CFG.K_MAX,
            tol=CFG.CONVERGENCE_TOL,
            de_seconds=CFG.DE_SECONDS
        )
        
        battery_manager = BatteryStateManager(M, em)
        
        # 测量多个时间步的运行时间
        T_test = 10
        slot_times = []
        
        for t in range(T_test):
            time_s = t * CFG.DE_SECONDS
            states = constellation.update(time_s)
            distances = constellation.compute_distances(time_s)
            illuminated = np.array([states[m]['illuminated'] for m in range(M)])
            sun_angles = np.array([states[m]['sun_angle'] for m in range(M)])
            p_max_dyn = battery_manager.get_dynamic_p_max(illuminated, sun_angles)
            
            start_time = time.perf_counter()
            
            result = algorithm.run(
                flows=flows,
                distances=distances,
                battery_levels=battery_manager.battery_levels.copy(),
                p_max_dynamic=p_max_dyn,
                C_min_j=em.C_min
            )
            
            end_time = time.perf_counter()
            slot_times.append(end_time - start_time)
            
            battery_manager.step(illuminated, sun_angles, result['P'])
        
        avg_time = np.mean(slot_times[2:])  # 跳过前2个热身
        runtimes[label] = avg_time
        
        print(f"{label}: M={M}, 平均时间={avg_time:.3f}s/slot")
    
    # ============================================================
    # 绘图
    # ============================================================
    fig, ax = plt.subplots(1, 1, figsize=(3.5, 2.5))
    
    n_sat_values = n_satellites_list
    runtime_values = [runtimes[label] for label in test_constellations.keys()]
    labels_short = ['Shell A\n(172)', 'Shell B\n(348)', 'Kuiper\n(784)']
    
    ax.bar(labels_short, runtime_values,
           color=['#1A4B8C', '#E05A2B', '#2CA02C'],
           width=0.5, alpha=0.85)
    
    # 添加O(n_t)参考线
    n_t_ref = np.array(n_sat_values)
    ref_scale = runtime_values[0] / n_sat_values[0]
    ref_values = ref_scale * n_t_ref
    ax.plot(labels_short, ref_values,
            'k--', linewidth=0.8, alpha=0.7,
            label=r'$\mathcal{O}(n_t)$ ref.')
    
    ax.set_ylabel('Runtime per slot (s)', fontsize=8)
    ax.set_xlabel('Constellation', fontsize=8)
    ax.tick_params(labelsize=7)
    ax.legend(fontsize=6)
    ax.grid(True, linestyle=':', alpha=0.5, linewidth=0.5, axis='y')
    
    plt.tight_layout()
    
    os.makedirs('results/figures', exist_ok=True)
    plt.savefig('results/figures/fig5_runtime.pdf',
                dpi=300, bbox_inches='tight')
    plt.savefig('results/figures/fig5_runtime.png',
                dpi=300, bbox_inches='tight')
    plt.close()
    
    print("Fig.5 已保存到 results/figures/fig5_runtime.pdf")
    
    # 打印运行时间表
    print("\n运行时间汇总：")
    for label, t in runtimes.items():
        print(f"  {label}: {t:.3f} s/slot")


if __name__ == '__main__':
    run_fig5()
```
